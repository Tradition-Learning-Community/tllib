cmake_minimum_required(VERSION 3.27)
project(tllib_binder_spike LANGUAGES CXX)
find_package(Python 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(nanobind CONFIG REQUIRED)
pybind11_add_module(_spike_pybind pybind.cpp)
nanobind_add_module(_spike_nanobind nanobind.cpp)
foreach(name IN ITEMS _spike_pybind _spike_nanobind)
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
  set_target_properties(${name} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
endforeach()
